            if (base.includes('.')) base = base.replace(/\.[^.]+$/, '');
        } catch (e) { base = ''; }
        if (!base) base = (img.alt || img.title || '').trim();
        if (!base) base = `img_${idx}`;
        base = base.replace(/[\\\/:*?"<>|]/g, '_').slice(0, 80);
        return base + '.' + ext;
    }

    doc.addEventListener('mousedown', e => {
        if (e.target.closest('.toolbar') || e.target.closest('.filterbar') || e.target.closest('#modalOverlay') || e.target.closest('#contextMenu')) return;
        if (e.target.closest('.editable')) return; // permite seleccionar texto sin disparar marquee/selección
        isDragging = true; didDrag = false;
        startX = e.pageX; startY = e.pageY;
    });

    doc.addEventListener('mousemove', e => {
        if (!isDragging) return;
        let dx = Math.abs(e.pageX - startX), dy = Math.abs(e.pageY - startY);
        if (!didDrag && (dx > DRAG_THRESHOLD || dy > DRAG_THRESHOLD)) {
            didDrag = true;
            if (!e.ctrlKey && !e.shiftKey) clearSelection();
            marquee.style.display = 'block';
        }
        if (!didDrag) return;
        let x = Math.min(startX, e.pageX), y = Math.min(startY, e.pageY);
        let w = Math.abs(startX - e.pageX), h = Math.abs(startY - e.pageY);
        marquee.style.left = x+'px'; marquee.style.top = y+'px';
        marquee.style.width = w+'px'; marquee.style.height = h+'px';
        items.forEach(el => {
            let r = el.getBoundingClientRect();
            let sY = newWin.scrollY;
            if (!(x > r.right || x+w < r.left || y > r.bottom+sY || y+h < r.top+sY)) el.classList.add('selected');
        });
    });

    doc.addEventListener('mouseup', e => {
        if (!isDragging) return;
        isDragging = false;
        marquee.style.display = 'none';
        if (didDrag) return;

        const checkEl = e.target.closest('.check');
        const target = e.target.closest('.thumb');
        if (!target) return;
        const idx = items.indexOf(target);
        setFocus(idx);

        if (checkEl) { target.classList.toggle('selected'); lastClickedIndex = idx; return; }

        if (e.shiftKey && lastClickedIndex !== null) {
            if (!e.ctrlKey) clearSelection();
            selectRange(lastClickedIndex, idx);
        } else if (e.ctrlKey) {
            target.classList.toggle('selected'); lastClickedIndex = idx;
        } else {
            clearSelection(); target.classList.add('selected'); lastClickedIndex = idx;
        }
    });

    const modalOverlay = doc.getElementById('modalOverlay');
    const modalViewport = doc.getElementById('modalViewport');
    const modalStage = doc.getElementById('modalStage');
    const modalImg = doc.getElementById('modalImg');
    const modalCanvas = doc.getElementById('modalCanvas');
    const mctx = modalCanvas.getContext('2d');
    const modalCaption = doc.getElementById('modalCaption');

    let modalIndex = 0;
    let zoom = 1, panX = 0, panY = 0;
    let pencilMode = false;
    let isPanning = false, panStartX, panStartY;
    let isDrawing = false;
    let undoStack = [];

    function applyTransform() {
        modalStage.style.transform = `translate(${panX}px, ${panY}px) scale(${zoom})`;
        doc.getElementById('modalZoomReset').textContent = Math.round(zoom * 100) + '%';
    }

    function setupCanvasForImage() {
        const it = items_data[modalIndex];
        const tmp = new Image();
        return new Promise(resolve => {
            tmp.onload = () => {
                modalCanvas.width = tmp.naturalWidth;
                modalCanvas.height = tmp.naturalHeight;
                mctx.clearRect(0, 0, modalCanvas.width, modalCanvas.height);
                resolve();
            };
            tmp.src = it.url;
        });
    }

    async function openModal(idx) {
        modalIndex = idx;
        const it = items_data[idx];
        modalImg.src = it.url;
        zoom = 1; panX = 0; panY = 0;
        applyTransform();
        await setupCanvasForImage();
        undoStack = [];
        modalCaption.textContent = `${it.name} · ${fmtSize(it.size)} · ${fmtDate(it.date)}  (${idx+1}/${items_data.length})`;
        modalOverlay.classList.add('open');
    }
    function closeModal() { modalOverlay.classList.remove('open'); }
    async function modalNav(delta) {
        modalIndex = (modalIndex + delta + items_data.length) % items_data.length;
        await openModal(modalIndex);
    }

    doc.getElementById('modalClose').onclick = closeModal;
    doc.getElementById('modalPrev').onclick = () => modalNav(-1);
    doc.getElementById('modalNext').onclick = () => modalNav(1);
    modalOverlay.addEventListener('click', e => { if (e.target === modalOverlay) closeModal(); });

    function setZoom(newZoom) {
        zoom = Math.min(8, Math.max(0.2, newZoom));
        applyTransform();
    }
    doc.getElementById('modalZoomIn').onclick = () => setZoom(zoom + 0.25);
    doc.getElementById('modalZoomOut').onclick = () => setZoom(zoom - 0.25);
    doc.getElementById('modalZoomReset').onclick = () => { zoom = 1; panX = 0; panY = 0; applyTransform(); };

    modalViewport.addEventListener('wheel', e => {
        if (!modalOverlay.classList.contains('open')) return;
        e.preventDefault();
        setZoom(zoom + (e.deltaY < 0 ? 0.15 : -0.15));
    }, { passive: false });

    modalViewport.addEventListener('mousedown', e => {
        if (pencilMode) return;
        isPanning = true;
        modalViewport.classList.add('panning');
        panStartX = e.clientX - panX;
        panStartY = e.clientY - panY;
    });
    doc.addEventListener('mousemove', e => {
        if (!isPanning) return;
        panX = e.clientX - panStartX;
        panY = e.clientY - panStartY;
        applyTransform();
    });
    doc.addEventListener('mouseup', () => {
        isPanning = false;
        modalViewport.classList.remove('panning');
    });

    const btnPencil = doc.getElementById('modalPencil');
    const colorPicker = doc.getElementById('modalColor');
    const brushSize = doc.getElementById('modalBrushSize');

    btnPencil.onclick = () => {
        pencilMode = !pencilMode;
        btnPencil.classList.toggle('active', pencilMode);
        modalViewport.classList.toggle('pencil-mode', pencilMode);
    };

    function pushUndoSnapshot() {
        undoStack.push(mctx.getImageData(0, 0, modalCanvas.width, modalCanvas.height));
        if (undoStack.length > 25) undoStack.shift();
    }
    function canvasPointFromEvent(e) {
        const rect = modalCanvas.getBoundingClientRect();
        const x = (e.clientX - rect.left) / rect.width * modalCanvas.width;
        const y = (e.clientY - rect.top) / rect.height * modalCanvas.height;
        return { x, y };
    }

    modalCanvas.addEventListener('mousedown', e => {
        if (!pencilMode) return;
        isDrawing = true;
        pushUndoSnapshot();
        const p = canvasPointFromEvent(e);
        mctx.beginPath();
        mctx.moveTo(p.x, p.y);
        mctx.lineCap = 'round';
        mctx.lineJoin = 'round';
        mctx.strokeStyle = colorPicker.value;
        mctx.lineWidth = parseInt(brushSize.value, 10) * (modalCanvas.width / modalCanvas.getBoundingClientRect().width);
    });
    doc.addEventListener('mousemove', e => {
        if (!isDrawing || !pencilMode) return;
        const p = canvasPointFromEvent(e);
        mctx.lineTo(p.x, p.y);
        mctx.stroke();
    });
    doc.addEventListener('mouseup', () => { isDrawing = false; });

    doc.getElementById('modalUndo').onclick = () => {
        if (!undoStack.length) return;
        mctx.putImageData(undoStack.pop(), 0, 0);
    };
    doc.getElementById('modalClear').onclick = () => {
        pushUndoSnapshot();
        mctx.clearRect(0, 0, modalCanvas.width, modalCanvas.height);
    };

    function reencodeDataUrl(dataUrl, mime, quality) {
        return new Promise(resolve => {
            const img = new Image();
            img.onload = () => {
                const c = doc.createElement('canvas');
                c.width = img.naturalWidth;
                c.height = img.naturalHeight;
                const cx = c.getContext('2d');
                if (mime === 'image/jpeg') {
                    cx.fillStyle = '#ffffff';
                    cx.fillRect(0, 0, c.width, c.height);
                }
                cx.drawImage(img, 0, 0);
                resolve(c.toDataURL(mime, quality));
            };
            img.src = dataUrl;
        });
    }
    const mimeFor = fmt => fmt === 'jpeg' ? 'image/jpeg' : fmt === 'png' ? 'image/png' : 'image/webp';
    const extFor  = fmt => fmt === 'jpeg' ? 'jpg' : fmt;

    function refreshThumbnail(idx) {
        const el = items[idx];
        if (!el) return;
        const it = items_data[idx];
        el.querySelector('.thumb-img img').src = it.url;
        el.querySelector('.thumb-name').textContent = it.name;
        const colType = el.querySelector('.col-type'); if (colType) colType.textContent = 'Archivo ' + it.ext.toUpperCase();
        const colSize = el.querySelector('.col-size'); if (colSize) colSize.textContent = fmtSize(it.size);
        const meta = el.querySelector('.thumb-meta'); if (meta) meta.textContent = `Archivo ${it.ext.toUpperCase()} · ${fmtSize(it.size)}`;
    }

    doc.getElementById('modalSave').onclick = () => {
        const it = items_data[modalIndex];
        const out = doc.createElement('canvas');
        out.width = modalCanvas.width;
        out.height = modalCanvas.height;
        const octx = out.getContext('2d');
        const baseImg = new Image();
        baseImg.onload = () => {
            octx.drawImage(baseImg, 0, 0, out.width, out.height);
            octx.drawImage(modalCanvas, 0, 0);
            const mime = mimeFor(it.ext === 'jpg' ? 'jpeg' : it.ext);
            const newUrl = out.toDataURL(mime, 0.95);
            it.url = newUrl;
            it.size = Math.round((newUrl.split(',')[1] || '').length * 0.75);
            modalImg.src = newUrl;
            mctx.clearRect(0, 0, modalCanvas.width, modalCanvas.height);
            undoStack = [];
            refreshThumbnail(modalIndex);
            modalCaption.textContent = `${it.name} · ${fmtSize(it.size)} · guardado ✓`;
        };
        baseImg.src = it.url;
    };

    doc.getElementById('modalConvert').onclick = async () => {
        const fmt = doc.getElementById('modalFormat').value;
        const it = items_data[modalIndex];
        const newUrl = await reencodeDataUrl(it.url, mimeFor(fmt), 0.95);
        it.url = newUrl;
        it.ext = extFor(fmt);
        it.name = it.name.replace(/\.[^.]+$/, '') + '.' + it.ext;
        it.size = Math.round((newUrl.split(',')[1] || '').length * 0.75);
        modalImg.src = newUrl;
        await setupCanvasForImage();
        refreshThumbnail(modalIndex);
        modalCaption.textContent = `${it.name} · ${fmtSize(it.size)} · convertido a ${fmt.toUpperCase()} ✓`;
    };

    doc.getElementById('btnConvertSel').onclick = async () => {
        const fmt = doc.getElementById('formatSelect').value;
        const selected = items.filter(el => el.classList.contains('selected'));
        for (const el of selected) {
            const idx = parseInt(el.dataset.index, 10);
            const it = items_data[idx];
            const newUrl = await reencodeDataUrl(it.url, mimeFor(fmt), 0.95);
            it.url = newUrl;
            it.ext = extFor(fmt);
            it.name = it.name.replace(/\.[^.]+$/, '') + '.' + it.ext;
            it.size = Math.round((newUrl.split(',')[1] || '').length * 0.75);
            refreshThumbnail(idx);
        }
    };

    // ---------- Teclado (galería + modal) ----------
    doc.addEventListener('keydown', e => {
        if (doc.activeElement && doc.activeElement.isContentEditable) {
            if (e.key === 'Enter') { e.preventDefault(); doc.activeElement.blur(); }
            if (e.key === 'Escape') { doc.activeElement.blur(); }
            return; // no interferir con la edición de texto
        }
        if (modalOverlay.classList.contains('open')) {
            if (e.key === 'Escape') closeModal();
            if (e.key === 'ArrowRight' && !pencilMode) modalNav(1);
            if (e.key === 'ArrowLeft' && !pencilMode) modalNav(-1);
            if (e.key === '+') setZoom(zoom + 0.25);
            if (e.key === '-') setZoom(zoom - 0.25);
            return;
        }
        if (!items.length || focusedIndex === null) return;

        const cols = getColumnCount();
        const key = e.key;
        const isArrow = ['ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(key);

        if (e.ctrlKey && key === ' ') {
            e.preventDefault();
            items[focusedIndex].classList.toggle('selected');
            lastClickedIndex = focusedIndex;
            return;
        }
        if (e.ctrlKey && (key === 'a' || key === 'A')) {
            e.preventDefault();
            items.forEach(el => el.classList.add('selected'));
            return;
        }
        if (key === 'Enter') { e.preventDefault(); openModal(focusedIndex); return; }
        if (!isArrow) return;
        e.preventDefault();

        let next = focusedIndex;
        if (key === 'ArrowRight') next = Math.min(focusedIndex + 1, items.length - 1);
        if (key === 'ArrowLeft')  next = Math.max(focusedIndex - 1, 0);
        if (key === 'ArrowDown')  next = Math.min(focusedIndex + cols, items.length - 1);
        if (key === 'ArrowUp')    next = Math.max(focusedIndex - cols, 0);

        if (e.ctrlKey && e.shiftKey) {
            selectRange(lastClickedIndex !== null ? lastClickedIndex : focusedIndex, next);
            setFocus(next);
        } else if (e.shiftKey) {
            if (lastClickedIndex === null) lastClickedIndex = focusedIndex;
            clearSelection();
            selectRange(lastClickedIndex, next);
            setFocus(next);
        } else {
            clearSelection();
            items[next].classList.add('selected');
            lastClickedIndex = next;
            setFocus(next);
        }
    });

        const max = parseFloat(doc.getElementById('filterSizeMax').value);

            const kb = it.size / 1024;
            const okMin = isNaN(min) || kb >= min;
            const okMax = isNaN(max) || kb <= max;
            if (okMin && okMax) items[i].classList.add('selected');
        });

        const minV = doc.getElementById('filterDateMin').value;
        const maxV = doc.getElementById('filterDateMax').value;
        const min = minV ? new Date(minV) : null;
        const max = maxV ? new Date(maxV) : null;
        clearSelection();
        items_data.forEach((it, i) => {
            const okMin = !min || it.date >= min;
            const okMax = !max || it.date <= max;
            if (okMin && okMax) items[i].classList.add('selected');
        });
    };

    // ---------- PDF con niveles de compresión ----------
    function compressForPrint(dataUrl, mode) {
        return new Promise(resolve => {
            if (mode === 'normal') { resolve(dataUrl); return; }
            const img = new Image();
            img.onload = () => {
                const scale = mode === 'extra' ? 0.5 : 0.75;
                const quality = mode === 'extra' ? 0.45 : 0.65;
                const c = doc.createElement('canvas');
                c.width = Math.max(1, Math.round(img.naturalWidth * scale));
                c.height = Math.max(1, Math.round(img.naturalHeight * scale));
                const cx = c.getContext('2d');
                cx.drawImage(img, 0, 0, c.width, c.height);
                resolve(c.toDataURL('image/jpeg', quality));
            };
            img.src = dataUrl;
        });
    }

    doc.getElementById('btnPDF').onclick = async () => {
        const mode = doc.getElementById('pdfQuality').value; // normal | compress | extra
        const printArea = doc.getElementById('printArea');
        printArea.innerHTML = '';
        const selected = items.filter(el => el.classList.contains('selected'));
        for (const el of selected) {
            const idx = el.dataset.index;
            const finalUrl = await compressForPrint(items_data[idx].url, mode);
            const page = doc.createElement('div');
            page.className = 'print-page';
            page.innerHTML = `<img src="${finalUrl}">`;
            printArea.appendChild(page);
        }
        newWin.print();
    };

    doc.getElementById('btnDL').onclick = async () => {
        const selected = items.filter(el => el.classList.contains('selected'));
        for (let el of selected) {
            const idx = el.dataset.index;
            const it = items_data[idx];
            let a = doc.createElement('a');
            a.href = it.url; a.download = it.name; a.click();
            await new Promise(r => setTimeout(r, 400));
        }
    };

let contextMenuVisible = false;

const columnMap = {
    'colName': '.thumb-name',
    'colDate': '.col-date',
    'colType': '.col-type',
    'colSize': '.col-size',
    'colTitle': '.col-title',
    'colTags': '.col-tags',
    'colUrl': '.col-url',
    'colDesc': '.col-desc',
    'colThumbs': '.thumb-img, .check'
};

function hideContextMenu() {
    if (!contextMenuVisible) return;
    contextMenuVisible = false;
    contextMenu.style.display = 'none';
    contextMenu.classList.remove('show');
}

function toggleColumn(colId, visible) {
    const selector = columnMap[colId];
    if (!selector) return;
    
    doc.querySelectorAll(selector).forEach(el => {
        el.style.display = visible ? '' : 'none';
    });
    
    const checkbox = doc.getElementById(colId);
    if (checkbox) checkbox.checked = visible;
}

// Click derecho en la galería
gallery.addEventListener('contextmenu', e => {
    e.preventDefault();
    e.stopPropagation();
    
    if (!gallery.classList.contains('view-details')) {
        alert('Cambia a vista "Detalles" para personalizar columnas');
        return;
    }
    
    // Si ya está visible, lo ocultamos
    if (contextMenuVisible) {
        hideContextMenu();
        return;
    }
    
    // Posicionar menú
    const x = Math.min(e.clientX, window.innerWidth - 240);
    const y = Math.min(e.clientY, window.innerHeight - 400);
    
    contextMenu.style.left = x + 'px';
    contextMenu.style.top = y + 'px';
    contextMenu.style.display = 'block';
    contextMenuVisible = true;
    
    // Pequeño delay para la animación
    setTimeout(() => contextMenu.classList.add('show'), 10);
});

// Prevenir que el menú contextual se abra sobre sí mismo
contextMenu.addEventListener('contextmenu', e => {
    e.preventDefault();
    e.stopPropagation();
});

// Cerrar menú al hacer click fuera
doc.addEventListener('mousedown', e => {
    if (!contextMenuVisible) return;
    if (e.target.closest('#contextMenu')) return;
    if (e.target.closest('.toolbar')) return;
    if (e.target.closest('.filterbar')) return;
    if (e.target.closest('#modalOverlay')) return;
    
    // Pequeño delay para evitar conflictos
    setTimeout(() => {
        if (contextMenuVisible && !e.target.closest('#contextMenu')) {
            hideContextMenu();
        }
    }, 10);
});

// Cerrar con ESC
doc.addEventListener('keydown', e => {
    if (e.key === 'Escape' && contextMenuVisible) {
        e.preventDefault();
        hideContextMenu();
    }
});

// Event listeners para checkboxes
Object.keys(columnMap).forEach(colId => {
    const checkbox = doc.getElementById(colId);
    if (!checkbox) return;
    
    checkbox.addEventListener('change', e => {
        e.stopPropagation();
        toggleColumn(colId, e.target.checked);
    });
    
    // Prevenir que los clicks en checkboxes cierren el menú
    checkbox.addEventListener('mousedown', e => {
        e.stopPropagation();
    });
});

// Inicializar columnas
function initColumnVisibility() {
    setTimeout(() => {
        Object.keys(columnMap).forEach(colId => {
            const checkbox = doc.getElementById(colId);
            if (checkbox && checkbox.checked) {
                toggleColumn(colId, true);
            }
        });
    }, 100);
}
